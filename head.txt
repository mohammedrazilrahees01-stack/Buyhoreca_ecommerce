async function shopifyFetch<T>(query:string,variables:Record<string,unknown>={}):Promise<T>{
  const controller=new AbortController();
  const timeout=window.setTimeout(()=>controller.abort(),20000);
  try{
    const r=await fetch("/api/shopify",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({query,variables}),cache:"no-store",signal:controller.signal});
    const j=await r.json().catch(()=>null);
    if(!r.ok||j?.errors?.length)throw new Error("Shopify request failed");
    return j.data;
  }catch(error){
    if(error instanceof DOMException&&error.name==="AbortError")throw new Error("Shopify request timed out");
    throw error;
  }finally{window.clearTimeout(timeout)}
}
